% --- Safety fallback for ORCID toggle ---
\@ifundefined{orcidbooltrue}{%
    \newif\iforcidbool
    \orcidbooltrue
}{}

% --- End-of-list sentinel ---
\def\@nnil{\@nil}

% --- Check if an item is blank/whitespace without \edef ---
\long\def\@checkitemblank#1{%
    \expandafter\if\expandafter\relax\detokenize\expandafter{\@gobble#1.}\relax
        \expandafter\@firstoftwo
    \else
        \expandafter\@secondoftwo
    \fi
}

% --- Count items in comma-separated list ---
\newcount\authorcount
\long\def\countauthorcovers#1{%
    \authorcount=0\relax
    \expandafter\@countauthors@i#1,\@nil,%
}
\long\def\@countauthors@i#1,{%
    \def\@curritem{#1}%
    \ifx\@curritem\@nnil
    \else
        \@checkitemblank{#1}{%
            % Skip blank / trailing comma entry
        }{%
            \advance\authorcount by 1\relax
        }%
        \expandafter\@countauthors@i
    \fi
}

% --- Helper: Generate footnote for Email & Affiliation ---
\long\def\@makeauthorfootnote#1#2{%
    \if\relax\detokenize{#1}\relax
        % Email is empty
        \if\relax\detokenize{#2}\relax
            % Affiliation is also empty -> emit nothing
        \else
            % Affiliation only
            \footnote{#2}%
        \fi
    \else
        % Email is present
        \if\relax\detokenize{#2}\relax
            % Email only
            \footnote{\href{mailto:#1}{\texttt{#1}}}%
        \else
            % Both Email and Affiliation
            \footnote{\href{mailto:#1}{\texttt{#1}}, #2}%
        \fi
    \fi
}

% --- Process single author: {ORCID}{Name}{Email}{Affiliation} ---
\long\def\@processentry#1#2#3#4{%
    \iforcidbool
        \if\relax\detokenize{#1}\relax
            \textsc{#2}%
        \else
            \href{https://orcid.org/#1}{\textsc{#2}%
            \hspace{0.5mm}\includegraphics[height=0.7em]{Resources/misc/orcid-ID.png}}%
        \fi
    \else
        \textsc{#2}%
    \fi
    \@makeauthorfootnote{#3}{#4}%
    \\%
}

% --- Process entire author list ---
\long\def\@processauthorlist#1{%
    \expandafter\@processauthorlist@i#1,\@nil,%
}
\long\def\@processauthorlist@i#1,{%
    \def\@curritem{#1}%
    \ifx\@curritem\@nnil
    \else
        \@checkitemblank{#1}{%
            % Skip blank / trailing comma entry
        }{%
            \@processentry#1%
        }%
        \expandafter\@processauthorlist@i
    \fi
}

% --- Print authors ---
\newcommand{\printauthors}{%
    \countauthorcovers{\coverauthors}%
    \ifnum\authorcount>0
        \begin{center}
        \large\textbf{%
            \ifnum\authorcount>1%
                \studenttextplural%
            \else%
                \studenttextsingular%
            \fi%
        }
        \end{center}\vspace{-0.35cm}%
        \@processauthorlist{\coverauthors}%
    \fi
}

% --- Print professors ---
\newcommand{\printprof}{%
    \countauthorcovers{\coverprofessor}%
    \ifnum\authorcount>0
        \begin{center}
        \large\textbf{%
            \ifnum\authorcount>1%
                \professortextplural%
            \else%
                \professortextsingular%
            \fi%
        }
        \end{center}\vspace{-0.35cm}%
        \@processauthorlist{\coverprofessor}%
    \fi
}

% --- Print assistants ---
\newcommand{\printprofassist}{%
    \countauthorcovers{\coverassist}%
    \ifnum\authorcount>0
        \begin{center}
        \large\textbf{%
            \ifnum\authorcount>1%
                \assisttextplural%
            \else%
                \assisttextsingular%
            \fi%
        }
        \end{center}\vspace{-0.35cm}%
        \@processauthorlist{\coverassist}%
    \fi
}
